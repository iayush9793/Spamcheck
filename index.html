let domainMap = new Map();
let emailRecords = [];

async function refreshSheet() {
  try {
    const res = await axios.get(SHEET_CSV_URL);
    const lines = res.data.split('\n');
    domainMap.clear();
    for (let i = 1; i < lines.length; i++) {
      const [dom, et] = lines[i].split(',').map(s => s ? s.trim().replace(/^"|"$/g, '') : '');
      if (dom && et) domainMap.set(dom.toLowerCase(), et);
    }
    console.log(`[Sheet Sync] Cached ${domainMap.size} domain mappings.`);
  } catch (err) {
    console.error('[Sheet Error]', err.message);
  }
}

async function syncMailboxes() {
  for (const acc of ACCOUNTS) {
    try {
      const tokenRes = await axios.post('https://oauth2.googleapis.com/token', {
        client_id: CLIENT_ID,
        client_secret: CLIENT_SECRET,
        refresh_token: acc.token,
        grant_type: 'refresh_token'
      });
      const token = tokenRes.data.access_token;
      const listRes = await axios.get('https://gmail.googleapis.com/gmail/v1/users/me/messages?maxResults=15&includeSpamTrash=true', {
        headers: { Authorization: `Bearer ${token}` }
      });
      console.log(`[${acc.name}] Fetched ${listRes.data.messages?.length || 0} messages`);
    } catch (e) {
      console.error(`[${acc.name}] Error:`, e.message);
    }
  }
}

setInterval(syncMailboxes, 30000);
refreshSheet();
syncMailboxes();

app.get('/api/records', (req, res) => res.json(emailRecords));
app.listen(5000, () => console.log('Monitoring server active on port 5000'));
